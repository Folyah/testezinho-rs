local function createDmcStartup(d)
    local B = {
        stage = "Waiting for the game", finished = false,
        cleanup = nil,
        error = nil,
    }
    function B.Status(text)
        B.stage = text
        if not B.finished then pcall(d.show, "Auto GK | Starting", text) end
    end
    function B.Load(path)
        B.Status("Loading " .. path)
        local object = d.resolve(path)
        local result, closed
        d.spawn(function()
            local values = table.pack(pcall(d.require, object))
            if not closed then result = values end
        end)
        local deadline = d.now() + 30
        while not result and d.now() < deadline do d.wait() end
        closed = true
        assert(result, "Timed out loading " .. path)
        assert(result[1], "Cannot load " .. path .. ": " .. tostring(result[2]))
        assert(type(result[2]) == "table", "Unexpected module result: " .. path)
        return result[2]
    end
    function B.Run(main)
        B.Status(B.stage)
        local ok, err = pcall(function() d.ready(); main(B) end)
        B.finished = true
        if ok then
            pcall(d.hide)
        else
            local message = tostring(err)
            local cleanup = B.cleanup
            if cleanup then
                local cleaned, released = pcall(cleanup)
                if not cleaned or released == false then message = message .. "\nCleanup incomplete; rejoin before retrying." end
            end
            B.error = message
            d.warn("[Auto GK] " .. message)
            pcall(d.show, "Auto GK | Failed", message)
        end
        return ok, err
    end
    return B
end

local NewState = {
    lastTackleAt = 0,
    lastDiveAt = 0,
}

local function getDirectionName(direction, camera)
    local flatDir = Vector3.new(direction.X, 0, direction.Z).Unit
    local camForward = Vector3.new(camera.CFrame.LookVector.X, 0, camera.CFrame.LookVector.Z).Unit
    local camRight = Vector3.new(-camForward.Z, 0, camForward.X)
    
    local forwardDot = flatDir:Dot(camForward)
    local rightDot = flatDir:Dot(camRight)
    
    if forwardDot > 0.7 then return "F"
    elseif forwardDot < -0.7 then return "B"
    elseif rightDot > 0.7 then return "R"
    elseif rightDot < -0.7 then return "L"
    elseif forwardDot > 0 and rightDot > 0 then return "RF"
    elseif forwardDot > 0 and rightDot < 0 then return "LF"
    elseif forwardDot < 0 and rightDot > 0 then return "RB"
    elseif forwardDot < 0 and rightDot < 0 then return "LB"
    end
    return "F"
end

local function getDefendedGoal()
    local goal = M.Practice.GetDefendedGoalPart()
    if goal then return goal end
    local team = M.Teams.GetActorTeamName(LocalPlayer)
    if not team then return nil end
    local map = workspace:FindFirstChild("Map")
    local data = map and map:FindFirstChild("Data")
    local home = data and data:FindFirstChild(team)
    return home and home:FindFirstChild("Goal")
end

local function isBallHeadingToGoal(ballState, goal)
    if not ballState or not goal then return false end
    local ballPos = ballState.Position
    local ballVel = ballState.Velocity
    local goalPos = goal.Position
    local toGoal = (goalPos - ballPos).Unit
    local ballDir = ballVel.Unit
    return toGoal:Dot(ballDir) > 0.7
end

local function getBallDistanceToGoal(ballState, goal)
    if not ballState or not goal then return math.huge end
    return (ballState.Position - goal.Position).Magnitude
end

local function canDive()
    if not LocalPlayer.Character then return false end
    local root = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    if os.clock() - NewState.lastDiveAt < 1.5 then return false end
    if M.Dive.IsDiveConstraintActive(root) then return false end
    if M.Locks.HasGoalkeeperDiveBlock(LocalPlayer, M.Renderer.GetVolleyLockUserId()) then return false end
    return true
end

local function executeDive(direction)
    if not canDive() then return false end
    local character = LocalPlayer.Character
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    local camera = workspace.CurrentCamera
    if not camera then return false end
    local success = pcall(function()
        M.Dive.Run(character, root, direction, camera.CFrame, {})
    end)
    if success then NewState.lastDiveAt = os.clock() end
    return success
end

local function findNearestOpponent()
    local character = LocalPlayer.Character
    if not character then return nil end
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local myTeam = LocalPlayer.Team
    local nearest = nil
    local nearestDist = Config.TackleRange
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Team ~= myTeam and player.Character then
            local otherRoot = player.Character:FindFirstChild("HumanoidRootPart")
            if otherRoot then
                local dist = (otherRoot.Position - root.Position).Magnitude
                if dist < nearestDist then nearestDist = dist; nearest = player end
            end
        end
    end
    return nearest, nearestDist
end

local function detectTrickShot(ballState)
    if not Config.DetectTrickShots then return false end
    if not ballState then return false end
    local velocity = ballState.Velocity
    local speed = velocity.Magnitude
    if speed < Config.TrickShotVelocityThreshold then return false end
    local goal = getDefendedGoal()
    if not goal then return false end
    local isHeadingToGoal = isBallHeadingToGoal(ballState, goal)
    return not isHeadingToGoal
end

local function autoTackle()
    if not Config.AutoTackle then return false end
    if os.clock() - NewState.lastTackleAt < Config.TackleCooldown then return false end
    local target, dist = findNearestOpponent()
    if target and dist < Config.TackleRange then
        local targetRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if targetRoot then
            local character = LocalPlayer.Character
            local root = character:FindFirstChild("HumanoidRootPart")
            if root then
                local direction = (targetRoot.Position - root.Position).Unit
                local directionName = getDirectionName(direction, workspace.CurrentCamera)
                if executeDive(directionName) then
                    NewState.lastTackleAt = os.clock()
                    return true
                end
            end
        end
    end
    return false
end

local function reactToOffTarget(ballState, goal)
    if not Config.ReactToOffTarget then return false end
    if not ballState or not goal then return false end
    local isHeadingToGoal = isBallHeadingToGoal(ballState, goal)
    local distanceToGoal = getBallDistanceToGoal(ballState, goal)
    if not isHeadingToGoal and distanceToGoal < Config.OffTargetMaxDistance then
        if Config.DetectTrickShots and detectTrickShot(ballState) then
            if distanceToGoal < 15 then
                local character = LocalPlayer.Character
                local root = character:FindFirstChild("HumanoidRootPart")
                if root then
                    local direction = (ballState.Position - root.Position).Unit
                    local directionName = getDirectionName(direction, workspace.CurrentCamera)
                    return executeDive(directionName)
                end
            end
        end
    end
    return false
end
